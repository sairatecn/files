### 第1题（16-21题）——优化的字符串匹配

```cpp
#include <iostream>
#include <string>
#include <vector>
using namespace std;

int f(const string &s, const string &t) {
    int n = s.length(), m = t.length();
    vector<int> shift(128, m + 1);
    int i, j;
    for (j = 0; j < m; j++)
        shift[t[j]] = m - j;
    for (i = 0; i <= n - m; i += shift[s[i + m]]) {
        j = 0;
        while (j < m && s[i + j] == t[j]) j++;
        if (j == m) return i;
    }
    return -1;
}

int main() {
    string a, b;
    cin >> a >> b;
    cout << f(a, b) << endl;
    return 0;
}
```

> 该程序实现的是优化后的字符串模式匹配（类似Boyer-Moore-Horspool算法）。

### 第2题（22-27题）——k进制基数排序

```cpp
#include <iostream>
#include <vector>
using namespace std;

const int K = 5;  // k进制

void radix_sort(vector<int> &a, int n) {
    int max_val = 0;
    for (int i = 0; i < n; ++i)
        max_val = max(max_val, a[i]);
    for (int exp = 1; max_val / exp > 0; exp *= K) {
        vector<int> cnt(K, 0);
        vector<int> tmp(n);
        for (int i = 0; i < n; ++i)
            cnt[(a[i] / exp) % K]++;
        for (int i = 1; i < K; ++i)
            cnt[i] += cnt[i - 1];
        for (int i = n - 1; i >= 0; --i) {
            int digit = (a[i] / exp) % K;
            tmp[--cnt[digit]] = a[i];
        }
        a = tmp;
    }
}

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; ++i)
        cin >> a[i];
    radix_sort(a, n);
    for (int x : a) cout << x << " ";
    return 0;
}
```

### 第3题（28-33题）——递推与位运算

```cpp
#include <iostream>
#include <vector>
using namespace std;

int n, k;

int solve() {
    vector<int> a(n);
    for (int i = 0; i < n; ++i)
        a[i] = i;
    int ans = 0;
    // 程序计算从0开始，每次以特定步长前进
    for (int i = 0; i < n; ++i) {
        int cnt = 0;
        int cur = i;
        while (cur != 0) {
            cnt++;
            cur = (cur + k) % n;
        }
        ans = max(ans, cnt);
    }
    return ans;
}

int main() {
    cin >> n >> k;
    cout << solve() << endl;
    return 0;
}
```

> 程序判断在一个大小为n的环上，每次跳k步，求最大循环长度。
