### 第1题（16-21题）——球体体积交

```cpp
#include <iostream>
#include <cmath>
using namespace std;

const double PI = acos(-1.0);

double sq(double x) { return x * x; }

double getVolume(double r, double d) {
    return 4.0 / 3.0 * PI * r * r * r - PI * d * d * (r - d / 3.0);
}

double solve(double x1, double y1, double z1, double r1,
             double x2, double y2, double z2, double r2) {
    double d = sqrt(sq(x2 - x1) + sq(y2 - y1) + sq(z2 - z1));
    double t = (sq(r1) - sq(r2) + sq(d)) / (2 * d);
    if (d >= r1 + r2) return 0;
    if (d + r1 <= r2) return 4.0 / 3.0 * PI * r1 * r1 * r1;
    if (d + r2 <= r1) return 4.0 / 3.0 * PI * r2 * r2 * r2;
    return getVolume(r1, r1 - t) + getVolume(r2, r2 - (d - t));
}

int main() {
    double x1, y1, z1, r1, x2, y2, z2, r2;
    cin >> x1 >> y1 >> z1 >> r1;
    cin >> x2 >> y2 >> z2 >> r2;
    cout << (int)solve(x1, y1, z1, r1, x2, y2, z2, r2) << endl;
    return 0;
}
```

### 第2题（22-27题）——最大子段和

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int solve1(vector<int> &a) {
    int n = a.size();
    // 算法1：动态规划求最大子段和
    int ans = a[0], sum = a[0];
    for (int i = 1; i < n; i++) {
        sum = max(sum + a[i], a[i]);
        ans = max(ans, sum);
    }
    return ans;
}

int solve2(vector<int> &a, int l, int r) {
    if (l + 1 == r) return max(a[l], 0);
    int mid = (l + r) / 2;
    int ans = max(solve2(a, l, mid), solve2(a, mid, r));
    int sum = 0, maxl = 0, maxr = 0;
    for (int i = mid - 1; i >= l; --i) {
        sum += a[i];
        maxl = max(maxl, sum);
    }
    sum = 0;
    for (int i = mid; i < r; ++i) {
        sum += a[i];
        maxr = max(maxr, sum);
    }
    return max(ans, maxl + maxr);
}

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; ++i)
        cin >> a[i];
    cout << solve1(a) << endl;
    cout << solve2(a, 0, n) << endl;
    return 0;
}
```

### 第3题（28-33题）——Base64编码

```cpp
#include <iostream>
#include <string>
using namespace std;

char table[65] = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

string encode(string s) {
    string ret = "";
    for (int i = 0; i < s.size(); i += 3) {
        int val = 0;
        for (int j = 0; j < 3; ++j) {
            val <<= 8;
            if (i + j < s.size())
                val |= (unsigned char)s[i + j];
        }
        for (int j = 0; j < 4; ++j) {
            if (i + j * 3 / 4 >= s.size())
                ret += '=';
            else
                ret += table[(val >> (6 * (3 - j))) & 0x3F];
        }
    }
    return ret;
}

string decode(string s) {
    string ret = "";
    int val = 0, cnt = 0;
    for (int i = 0; i < s.size(); ++i) {
        if (s[i] == '=') break;
        char *p = strchr(table, s[i]);
        if (p) {
            val = (val << 6) | (p - table);
            cnt++;
            if (cnt == 4) {
                for (int j = 0; j < 3; ++j)
                    ret += (char)((val >> (8 * (2 - j))) & 0xFF);
                val = 0;
                cnt = 0;
            }
        }
    }
    return ret;
}

int main() {
    string s;
    cin >> s;
    cout << encode(s) << endl;
    cout << decode(encode(s)) << endl;
    return 0;
}
```

### 第2题（39-43题）RMQ问题（笛卡尔树+分块）

给定一个长度为n的数组a[1..n]（1≤n≤10⁵），有q（1≤q≤10⁵）个询问，每个询问给出l和r，询问a[l..r]中的最小值。

以下程序使用笛卡尔树+分块的方法求解RMQ问题。

```cpp
#include <iostream>
#include <cmath>
#include <algorithm>
using namespace std;

const int MAXN = 100005;
const int MAXB = 400;

int n, q;
int a[MAXN];
int ls[MAXN], rs[MAXN];
int stk[MAXN], top;
int dep[MAXN], euler[MAXN * 2], pos[MAXN], tot;
int block[MAXN * 2], B, bsize;
int min_val[MAXB][MAXB];
int prec[MAXB][MAXB];

void dfs(int u) {
    pos[u] = tot;
    euler[tot++] = u;
    if (ls[u]) {
        dep[ls[u]] = dep[u] + 1;
        dfs(ls[u]);
        euler[tot++] = u;
    }
    if (rs[u]) {
        dep[rs[u]] = dep[u] + 1;
        dfs(rs[u]);
        euler[tot++] = u;
    }
}

int ask(int l, int r) {
    if (l > r) swap(l, r);
    int bl = l / bsize, br = r / bsize;
    if (bl == br) {
        int ans = l;
        for (int i = l + 1; i <= r; ++i)
            if (dep[euler[i]] < dep[euler[ans]])
                ans = i;
        return ans;
    }
    int ans = min_val[bl + 1][br - 1];
    for (int i = l; i < (bl + 1) * bsize; ++i)
        if (dep[euler[i]] < dep[euler[ans]])
            ans = i;
    for (int i = br * bsize; i <= r; ++i)
        if (dep[euler[i]] < dep[euler[ans]])
            ans = i;
    return ans;
}

int main() {
    cin >> n;
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
        // 构建笛卡尔树
        int k = top;
        while (k > 0 && a[stk[k]] > a[i]) k--;
        if (k) ___①___ = i;
        if (k < top) ___②___ = stk[k + 1];
        stk[++k] = i;
        top = k;
    }
    // 求欧拉序
    dfs(stk[1]);
    // 分块
    B = sqrt(tot);
    bsize = tot / B;
    for (int i = 0; i < tot; ++i)
        block[i] = i / bsize;
    for (int i = 0; i < B; ++i) {
        int l = i * bsize, r = min(l + bsize - 1, tot - 1);
        min_val[i][i] = l;
        for (int j = l + 1; j <= r; ++j)
            if (dep[euler[j]] < dep[euler[min_val[i][i]]])
                min_val[i][i] = j;
        for (int j = i + 1; j < B; ++j)
            if (dep[euler[min_val[i][j - 1]]]
                < dep[euler[min_val[j][j]]])
                min_val[i][j] = min_val[i][j - 1];
            else
                min_val[i][j] = min_val[j][j];
    }
    // 处理询问
    cin >> q;
    while (q--) {
        int l, r;
        cin >> l >> r;
        int p = ask(pos[l], pos[r]);
        cout << a[euler[p]] << endl;
    }
    return 0;
}
```
